Backtracking
🌍 Real-World Analogy
A maze: try one path, hit a dead end? Go back and try the next path. You explore ALL options by making a choice, going deeper, undoing the choice, and trying the next one. It's brute force made smart with pruning.
🧠 When To Recognize
generate all combinationspermutationssubsetsN-QueensSudoku solverword search
💻 Code
// Subsets (LC #78)
public List<List<Integer>> subsets(int[] nums) {
    List<List<Integer>> res = new ArrayList<>();
    backtrack(nums, 0, new ArrayList<>(), res);
    return res;
}
void backtrack(int[] nums, int start, List<Integer> curr, List<List<Integer>> res) {
    res.add(new ArrayList<>(curr));
    for (int i = start; i < nums.length; i++) {
        curr.add(nums[i]);           // choose
        backtrack(nums, i+1, curr, res); // explore
        curr.remove(curr.size()-1);  // un-choose
    }
}

func subsets(nums []int) [][]int {
    res := [][]int{}
    var bt func(int, []int)
    bt = func(start int, curr []int) {
        tmp := make([]int, len(curr)); copy(tmp, curr)
        res = append(res, tmp)
        for i := start; i < len(nums); i++ {
            bt(i+1, append(curr, nums[i]))
        }
    }
    bt(0, []int{})
    return res
}

def subsets(nums):
    res = []
    def backtrack(start, curr):
        res.append(curr[:])
        for i in range(start, len(nums)):
            curr.append(nums[i])      # choose
            backtrack(i + 1, curr)     # explore
            curr.pop()                 # un-choose
    backtrack(0, [])
    return res

Trie (Prefix Tree)
🌍 Real-World Analogy
Phone autocomplete — as you type "app", the system narrows down to "apple", "application", "apply". A trie stores words as a tree of characters, with shared prefixes stored once.
💻 Code
// Trie (LC #208)
class Trie {
    Trie[] children = new Trie[26];
    boolean isEnd = false;
    void insert(String w) {
        Trie node = this;
        for (char c : w.toCharArray()) {
            if (node.children[c-'a'] == null) node.children[c-'a'] = new Trie();
            node = node.children[c-'a'];
        }
        node.isEnd = true;
    }
    boolean search(String w) {
        Trie node = find(w); return node != null && node.isEnd;
    }
    boolean startsWith(String p) { return find(p) != null; }
    Trie find(String s) {
        Trie node = this;
        for (char c : s.toCharArray()) {
            if (node.children[c-'a'] == null) return null;
            node = node.children[c-'a'];
        }
        return node;
    }
}

type Trie struct{ children [26]*Trie; isEnd bool }
func (t *Trie) Insert(w string) {
    node := t
    for _, c := range w {
        i := c-'a'; if node.children[i]==nil{node.children[i]=&Trie{}}
        node = node.children[i]
    }
    node.isEnd = true
}
func (t *Trie) Search(w string) bool { n:=t.find(w); return n!=nil&&n.isEnd }
func (t *Trie) StartsWith(p string) bool { return t.find(p)!=nil }
func (t *Trie) find(s string) *Trie {
    node := t
    for _, c := range s { i:=c-'a'; if node.children[i]==nil{return nil}; node=node.children[i] }
    return node
}

class TrieNode:
    def __init__(self):
        self.children = {}; self.is_end = False

class Trie:
    def __init__(self): self.root = TrieNode()
    def insert(self, word):
        node = self.root
        for c in word:
            if c not in node.children: node.children[c] = TrieNode()
            node = node.children[c]
        node.is_end = True
    def search(self, word):
        node = self._find(word); return node is not None and node.is_end
    def starts_with(self, prefix): return self._find(prefix) is not None
    def _find(self, s):
        node = self.root
        for c in s:
            if c not in node.children: return None
            node = node.children[c]
        return node

Heap / Top-K
🌍 Real-World Analogy
A VIP list — you only keep the top K most important people. When a new person arrives, if they outrank the least important VIP, they swap in. A min-heap of size K does this efficiently.
💻 Code
// Top K Frequent Elements (LC #347)
public int[] topKFrequent(int[] nums, int k) {
    Map<Integer,Integer> freq = new HashMap<>();
    for (int n : nums) freq.merge(n, 1, Integer::sum);
    PriorityQueue<Integer> pq = new PriorityQueue<>((a,b) -> freq.get(a)-freq.get(b));
    for (int key : freq.keySet()) {
        pq.offer(key);
        if (pq.size() > k) pq.poll();
    }
    return pq.stream().mapToInt(i->i).toArray();
}

// Use container/heap or sort approach
func topKFrequent(nums []int, k int) []int {
    freq := map[int]int{}
    for _, n := range nums { freq[n]++ }
    keys := make([]int, 0, len(freq))
    for key := range freq { keys = append(keys, key) }
    sort.Slice(keys, func(i, j int) bool { return freq[keys[i]]>freq[keys[j]] })
    return keys[:k]
}

import heapq
from collections import Counter
def top_k_frequent(nums, k):
    freq = Counter(nums)
    return [x for x, _ in heapq.nlargest(k, freq.items(), key=lambda x: x[1])]

Two Heaps (Median Finding)
🌍 Real-World Analogy
Imagine splitting a class by height into a "shorter half" and a "taller half". The median is between the tallest of the short group and the shortest of the tall group. A max-heap tracks the shorter half, a min-heap tracks the taller half.
💻 Code
// Find Median from Data Stream (LC #295)
PriorityQueue<Integer> lo = new PriorityQueue<>(Collections.reverseOrder()); // max-heap (smaller half)
PriorityQueue<Integer> hi = new PriorityQueue<>(); // min-heap (larger half)
public void addNum(int num) {
    lo.offer(num); hi.offer(lo.poll());
    if (lo.size() < hi.size()) lo.offer(hi.poll());
}
public double findMedian() {
    return lo.size() > hi.size() ? lo.peek() : (lo.peek() + hi.peek()) / 2.0;
}

// Requires custom heap implementation in Go
// lo = max-heap (smaller half), hi = min-heap (larger half)
// addNum: push to lo, move top to hi, rebalance

import heapq
class MedianFinder:
    def __init__(self):
        self.lo = []   # max-heap (negate values)
        self.hi = []   # min-heap
    def addNum(self, num):
        heapq.heappush(self.lo, -num)
        heapq.heappush(self.hi, -heapq.heappop(self.lo))
        if len(self.lo) < len(self.hi):
            heapq.heappush(self.lo, -heapq.heappop(self.hi))
    def findMedian(self):
        if len(self.lo) > len(self.hi): return -self.lo[0]
        return (-self.lo[0] + self.hi[0]) / 2
